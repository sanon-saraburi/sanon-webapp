// ─── Supabase REST Helpers ───────────────────────────────────────────
async function fetchOne(table: string, id: string, select: string): Promise<Record<string, string>> {
  try {
    const res = await fetch(
      `${SUPABASE_URL}/rest/v1/${table}?id=eq.${encodeURIComponent(id)}&select=${select}&limit=1`,
      { headers: { "apikey": SUPABASE_KEY, "Authorization": `Bearer ${SUPABASE_KEY}` } }
    );
    const data = await res.json();
    return Array.isArray(data) && data.length > 0 ? data[0] : {};
  } catch {
    return {};
  }
}

// ─── Helpers ────────────────────────────────────────────────────────
function fmtNum(v: unknown, digits = 2) {
  const n = Number(v);
  return isNaN(n) ? "-" : n.toLocaleString("th-TH", { minimumFractionDigits: digits, maximumFractionDigits: digits });
}

function fmtDate(v: unknown) {
  if (!v) return "-";
  const d = new Date(String(v));
  if (isNaN(d.getTime())) return String(v);
  return d.toLocaleDateString("th-TH", { year: "numeric", month: "long", day: "numeric" });
}

function row(icon: string, text: string) {

    type: "box", layout: "baseline", spacing: "sm",
    contents: [

      { type: "text", text: text, size: "sm", color: "#333333", wrap: true },
    ],
  };
}

// ─── Send LINE Push ──────────────────────────────────────────────────
async function sendLine(flexBubble: unknown, groupId?: string) {
  const toGroup = groupId || LINE_GROUP_ID;
  if (!toGroup) throw new Error("LINE_GROUP_ID not configured");

  const res = await fetch("https://api.line.me/v2/bot/message/push", {
    method: "POST",
    headers: { Authorization: `Bearer ${LINE_TOKEN}`, "Content-Type": "application/json" },
    body: JSON.stringify({
      to: toGroup,
      messages: [{ type: "flex", altText: "มีรายการรออนุมัติ — สานนท์", contents: flexBubble }],
    }),
  });
  if (!res.ok) {
    const err = await res.text();
    console.error("LINE API error:", res.status, err);
    throw new Error(`LINE push failed: ${res.status}`);
  }
}

// ─── CORS Headers ────────────────────────────────────────────────────
const CORS = {
  "Access-Control-Allow-Origin":  "*",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type, Authorization",
};

// ─── Main Handler ────────────────────────────────────────────────────
serve(async (req) => {
  // รับ CORS Preflight (browser ส่งมาก่อน POST จริงเสมอเมื่อ Content-Type: application/json)
  if (req.method === "OPTIONS") {
    return new Response("ok", { status: 200, headers: CORS });
  }

  if (!LINE_TOKEN || !LINE_GROUP_ID) {
    console.error("Missing LINE_CHANNEL_TOKEN or LINE_GROUP_ID");
    return new Response("config_missing", { status: 500, headers: CORS });
  }

  // ใช้ req.text() + JSON.parse() เพื่อกัน parse error และ log body จริงได้
  let rawBody = "";
  let body: { type: string; table: string; record: Record<string, unknown> };
  try {
    rawBody = await req.text();
    console.log("raw body:", rawBody.slice(0, 300));
    body = JSON.parse(rawBody);
  } catch (e) {
    console.error("parse error:", e, "raw:", rawBody.slice(0, 300));
    return new Response("parse_error", { status: 400, headers: CORS });
  }

  const { type, table, record } = body;

  if (type !== "INSERT") return new Response("skip:not_insert", { status: 200, headers: CORS });

  try {
    // ── System 1: Production ──────────────────────────────────────
    if (PROD_TABLES.includes(table)) {
      if (record.status !== "pending") return new Response("skip:not_pending", { status: 200, headers: CORS });
      // ใช้ recorder_name จาก record ถ้ามี (frontend ส่งมา) หรือ lookup จาก DB
      let recorderName = String(record.recorder_name ?? "");
      if (!recorderName && record.created_by) {
        const user = await fetchOne("app_users", String(record.created_by), "full_name");
        recorderName = user.full_name ?? "-";
      }
      if (!recorderName) recorderName = "-";
      await sendLine(buildProductionCard(table, record, recorderName));
      return new Response("sent:production", { status: 200, headers: CORS });
    }

    // ── System 2: Inventory ───────────────────────────────────────
    if (table === "inventory_transactions") {
      if (record.transaction_type !== "withdraw" || record.status !== "pending") {
        return new Response("skip:not_pending_withdraw", { status: 200, headers: CORS });
      }
      // ใช้ item_name/item_unit จาก record ถ้ามี (frontend ส่งมา) หรือ lookup จาก DB
      let itemName = String(record.item_name ?? "");
      let itemUnit = String(record.item_unit ?? "");
      if (!itemName && record.item_id) {
        const item = await fetchOne("inventory_items", String(record.item_id), "name,unit");
        itemName = item.name ?? "-";
        itemUnit = item.unit ?? "";
      }
      if (!itemName) itemName = "-";
      await sendLine(buildInventoryCard(record, itemName, itemUnit));
      return new Response("sent:inventory", { status: 200, headers: CORS });
    }

    // ── System 3: PM Repair ───────────────────────────────────────
    if (table === "pm_repair_logs") {
      await sendLine(buildRepairCard(record));
      return new Response("sent:pm_repair", { status: 200, headers: CORS });
    }

    // ── System 3: PM Meter Log ────────────────────────────────────
    if (table === "pm_meter_logs") {
      await sendLine(buildMeterLogCard(record));
      return new Response("sent:pm_meter", { status: 200, headers: CORS });
    }

    // ── System 3: PM Due Alert (overdue / near threshold) ─────────
    if (table === "pm_alert") {
      const alerts = record.alerts as unknown[];
      if (!alerts || alerts.length === 0) return new Response("skip:no_alerts", { status: 200, headers: CORS });
      await sendLine(buildPmAlertCard(record));
      return new Response("sent:pm_alert", { status: 200, headers: CORS });
    }

    // ── System 2: Low Stock Alert (real-time, หลังอนุมัติ/เบิก) ──
    if (table === "low_stock_alert") {
      await sendLine(buildLowStockCard(record));
      return new Response("sent:low_stock", { status: 200, headers: CORS });
    }

    // ── System 6: Leave Request → กลุ่ม HR ───────────────────────
    if (table === "leave_requests") {
      if (record.status !== "pending") return new Response("skip:not_pending", { status: 200, headers: CORS });
      await sendLine(buildLeaveRequestCard(record), LINE_GROUP_ID_HR || LINE_GROUP_ID);
      return new Response("sent:leave_request", { status: 200, headers: CORS });
    }

    // ── System 6: Pass Request → กลุ่ม HR ────────────────────────
    if (table === "pass_requests") {
      if (record.status === "pending") {
        await sendLine(buildPassRequestCard(record, true), LINE_GROUP_ID_HR || LINE_GROUP_ID);
        return new Response("sent:pass_pending", { status: 200, headers: CORS });
      }
      if (record.status === "approved") {
        await sendLine(buildPassRequestCard(record, false), LINE_GROUP_ID_HR || LINE_GROUP_ID);
        return new Response("sent:pass_walkin", { status: 200, headers: CORS });
      }
      return new Response("skip:pass_other_status", { status: 200, headers: CORS });
    }

    return new Response("skip:unknown_table", { status: 200, headers: CORS });

  } catch (err) {
    console.error("Error:", err);
    return new Response(String(err), { status: 500, headers: CORS });
  }
});
